# Daisin Open - web installer / updater (per user, no admin)
# 実行方法（Win+R または PowerShell）:
#   powershell -NoProfile -ExecutionPolicy Bypass -Command "irm https://daisin-e.github.io/daisin-trello-dashboard/DaisinOpen/install.txt | iex"
$ErrorActionPreference = 'Stop'
[Net.ServicePointManager]::SecurityProtocol = [Net.SecurityProtocolType]::Tls12
$Base = 'https://daisin-e.github.io/daisin-trello-dashboard/DaisinOpen'
$Dst  = Join-Path $env:LOCALAPPDATA 'DaisinOpen'
if (-not (Test-Path $Dst)) { New-Item -ItemType Directory -Path $Dst | Out-Null }

foreach ($f in 'daisin-open.ps1', 'daisin-link.ps1') {
  $resp = Invoke-WebRequest -Uri ("$Base/$f`?v=" + (Get-Date -Format yyyyMMddHHmmss)) -UseBasicParsing
  $bytes = $resp.Content; if ($bytes -is [string]) { $bytes = [Text.Encoding]::UTF8.GetBytes($bytes) }
  $text = (New-Object Text.UTF8Encoding $false).GetString($bytes).TrimStart([char]0xFEFF)
  # Windows PowerShell 5.1 が日本語を正しく読めるよう UTF-8 BOM 付きで保存
  [IO.File]::WriteAllText((Join-Path $Dst $f), $text, (New-Object Text.UTF8Encoding $true))
}

$ps = "powershell.exe -NoProfile -ExecutionPolicy Bypass -WindowStyle Hidden -File `"$Dst\{0}`" `"%1`""
$openCmd = $ps -f 'daisin-open.ps1'
$linkCmd = $ps -f 'daisin-link.ps1'

# 1) daisin-open: プロトコル（SharePoint のファイル/フォルダをデスクトップアプリ/エクスプローラーで開く）
$k = 'HKCU:\Software\Classes\daisin-open'
New-Item -Path $k -Force | Out-Null
Set-ItemProperty -Path $k -Name '(default)' -Value 'URL:Daisin Open Protocol'
Set-ItemProperty -Path $k -Name 'URL Protocol' -Value ''
New-Item -Path "$k\DefaultIcon" -Force | Out-Null
Set-ItemProperty -Path "$k\DefaultIcon" -Name '(default)' -Value "$env:SystemRoot\System32\shell32.dll,1"
New-Item -Path "$k\shell\open\command" -Force | Out-Null
Set-ItemProperty -Path "$k\shell\open\command" -Name '(default)' -Value $openCmd

# 2) 右クリックメニュー「Trello 用リンクをコピー」（ファイル用）
$m = 'HKCU:\Software\Classes\*\shell\DaisinLink'
New-Item -Path $m -Force | Out-Null
Set-ItemProperty -Path $m -Name '(default)' -Value 'Trello 用リンクをコピー'
Set-ItemProperty -Path $m -Name 'Icon' -Value "$env:SystemRoot\System32\shell32.dll,134"
New-Item -Path "$m\command" -Force | Out-Null
Set-ItemProperty -Path "$m\command" -Name '(default)' -Value $linkCmd

# 3) 右クリックメニュー「Trello 用リンク（URLのみ）をコピー」（ファイル用）
$m2 = 'HKCU:\Software\Classes\*\shell\DaisinLinkUrl'
New-Item -Path $m2 -Force | Out-Null
Set-ItemProperty -Path $m2 -Name '(default)' -Value 'Trello 用リンク（URLのみ）をコピー'
Set-ItemProperty -Path $m2 -Name 'Icon' -Value "$env:SystemRoot\System32\shell32.dll,134"
New-Item -Path "$m2\command" -Force | Out-Null
Set-ItemProperty -Path "$m2\command" -Name '(default)' -Value ($linkCmd + ' -Format url')

# ★ v2 追加: 右クリックメニュー「Trello 用リンクをコピー」（フォルダ用）
$mF = 'HKCU:\Software\Classes\Directory\shell\DaisinLink'
New-Item -Path $mF -Force | Out-Null
Set-ItemProperty -Path $mF -Name '(default)' -Value 'Trello 用リンクをコピー'
Set-ItemProperty -Path $mF -Name 'Icon' -Value "$env:SystemRoot\System32\shell32.dll,4"
New-Item -Path "$mF\command" -Force | Out-Null
Set-ItemProperty -Path "$mF\command" -Name '(default)' -Value $linkCmd

# ★ v2 追加: 右クリックメニュー「Trello 用リンク（URLのみ）をコピー」（フォルダ用）
$mFU = 'HKCU:\Software\Classes\Directory\shell\DaisinLinkUrl'
New-Item -Path $mFU -Force | Out-Null
Set-ItemProperty -Path $mFU -Name '(default)' -Value 'Trello 用リンク（URLのみ）をコピー'
Set-ItemProperty -Path $mFU -Name 'Icon' -Value "$env:SystemRoot\System32\shell32.dll,4"
New-Item -Path "$mFU\command" -Force | Out-Null
Set-ItemProperty -Path "$mFU\command" -Name '(default)' -Value ($linkCmd + ' -Format url')

# CUI 完結 (PowerShell 5.1 / 7 両対応, WinForms 非使用)
Write-Host ""
Write-Host "========================================" -ForegroundColor Green
Write-Host "  Daisin Open v2 install/update: DONE" -ForegroundColor Green
Write-Host "========================================" -ForegroundColor Green
Write-Host ""
Write-Host "- Shift+Right-click on files/folders in OneDrive" -ForegroundColor White
Write-Host "  -> 'Trello link copy' is available" -ForegroundColor Yellow
Write-Host "- Clicking Trello links opens apps/explorer directly" -ForegroundColor White
Write-Host ""
Write-Host "* v2 NEW: folder links are now supported" -ForegroundColor Cyan
Write-Host ""
Write-Host "Install path: $Dst" -ForegroundColor DarkGray
Write-Host ""
